  async function findHumanAudio(wordText) {
    const word = String(wordText || '').trim().toLowerCase();
    if (!word) return '';
    try {
      const q = encodeURIComponent(`${word} pronunciation`);
      const url = `https://commons.wikimedia.org/w/api.php?action=query&generator=search&gsrsearch=${q}&gsrnamespace=6&gsrlimit=12&prop=imageinfo&iiprop=url|mime&format=json&origin=*`;
      const controller = new AbortController();
      const timer = setTimeout(()=>controller.abort(), 4500);
      const res = await fetch(url, {signal:controller.signal});
      clearTimeout(timer);
      if (!res.ok) return '';
      const data = await res.json();
      const pages = Object.values(data.query?.pages || {}).filter(p => p.imageinfo?.[0]?.url);
      const scored = pages.map(p => {
        const title=(p.title||'').toLowerCase();
        const info=p.imageinfo[0];
        let score=0;
        if (title.includes(word)) score+=8;
        if (/en[-_ ]?(us|uk|gb)|english/.test(title)) score+=5;
        if (/pronunciation|pronounce/.test(title)) score+=3;
        if (/\.(ogg|oga|mp3|wav)$/i.test(info.url||'')) score+=3;
        if (/song|music|sentence|phrase/.test(title)) score-=6;
        return {url:info.url,score,title};
      }).sort((a,b)=>b.score-a.score);
      return scored[0]?.score >= 8 ? scored[0].url : '';
    } catch { return ''; }
  }

  async function resolveHumanAudioForWord(word, {announce=false}={}) {
    if (!word || word.pronunciationUrl) return false;
    const found = await findHumanAudio(word.word);
    if (!found) { if (announce) toast('No clear human recording found. Device voice will be used.'); return false; }
    word.pronunciationUrl = found;
    word.pronunciationSource = 'Wikimedia Commons';
    saveStateQuietly();
    if (currentView === 'parent') renderLibrary();
    if (announce) toast('Human pronunciation found on Wikimedia Commons.');
    return true;
  }
