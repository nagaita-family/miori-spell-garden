pronunciationUrlInput').parentElement.after(findAudio);
    if(w){
      let reset=$('#resetWordLearningBtn');if(!reset){reset=document.createElement('button');reset.type='button';reset.id='resetWordLearningBtn';reset.className='danger-link';reset.textContent='Reset Learning for This Word';reset.onclick=()=>resetWordLearning($('#editWordId').value);$('.modal-actions').prepend(reset)}reset.classList.remove('hidden');

    }else {$('#resetWordLearningBtn')?.classList.add('hidden');$('#deleteWordBtn')?.classList.add('hidden')}

  function closeWordModal(){$('#wordModal').classList.add('hidden')}

  function saveWordFromForm(e){
    e.preventDefault();const id=$('#editWordId').value;const wordText=$('#wordInput').value.trim().toLowerCase();if(!wordText)return;
    let w=id?state.library.find(x=>x.id===id):state.library.find(x=>x.word===wordText);
    const isNew=!w;if(!w)w=normalizeWord({id:uid(),word:wordText});
    const previousMiori=w.mioriSpelling||'';
    w.word=wordText;w.meaningEn=$('#meaningEnInput').value.trim();w.meaningJa=$('#meaningJaInput').value.trim();w.example=$('#exampleInput').value.trim();w.phonicsFocus=$('#phonicsInput').value.trim();w.pictureCue=$('#pictureCueInput').value.trim();w.mioriSpelling=$('#mioriSpellingInput').value.trim().toLowerCase();w.pronunciationUrl=$('#pronunciationUrlInput').value.trim();w.pronunciationSource=w.pronunciationUrl?'manual':'device';
    if (w.mioriSpelling!==previousMiori) w.learning.mioriWeakness = w.mioriSpelling && w.mioriSpelling!==w.word ? 3 : 0;
    if(isNew)state.library.push(w);const wantWeek=$('#thisWeekCheck').checked;if(wantWeek&&!state.thisWeek.includes(w.id))state.thisWeek.push(w.id);if(!wantWeek)state.thisWeek=state.thisWeek.filter(x=>x!==w.id);saveState();closeWordModal();renderParent();toast(isNew?'Word added.':'Word saved.');
  }

  function resetWordLearning(id){
    const w=state.library.find(x=>x.id===id);if(!w)return;if(!confirm(`Reset learning data for “${w.word}” only?`))return;w.learning=normalizeWord({word:w.word}).learning;saveState();closeWordModal();renderParent();toast('Learning data reset for this word.');
  }

  function deleteWord(id){
    const w=state.library.find(x=>x.id===id);if(!w)return;if(!confirm(`Delete “${w.word}” from the Word Library?`))return;
    state.library=state.library.filter(x=>x.id!==id);state.thisWeek=state.thisWeek.filter(x=>x!==id);saveState();closeWordModal();renderParent();toast('Word deleted.');
  }

  async function importWordPack(file){
    if(!file)return;try{
      const text=await file.text();const data=JSON.parse(text);const words=Array.isArray(data)?data:(Array.isArray(data.words)?data.words:[]);if(!words.length)throw new Error('No words found');
      if(data.weekName||data.title)state.weekName=data.weekName||data.title;
      const newWeek=[];
      for(const raw of words){if(!raw.word)continue;const key=String(raw.word).trim().toLowerCase();let existing=state.library.find(w=>w.word===key);if(existing){
          const learning=existing.learning;const id=existing.id;existing={...existing,...normalizeWord({...raw,id}),learning};Object.assign(state.library.find(w=>w.id===id),existing);newWeek.push(id);
        }else{const nw=normalizeWord(raw);state.library.push(nw);newWeek.push(nw.id)}}
      state.thisWeek=[...new Set(newWeek)];saveState();renderParent();toast(`Imported ${newWeek.length} words.`);
      const unresolved=newWeek.map(id=>state.library.find(w=>w.id===id)).filter(w=>w&&!w.pronunciationUrl);
      (async()=>{for(const w of unresolved) await resolveHumanAudioForWord(w); saveStateQuietly(); if(currentView==='parent')renderLibrary();})().catch(()=>{});
    }catch(e){console.error(e);toast('Could not import this Word Pack. Check the JSON format.',true)}finally{$('#importInput').value=''}
  }

  function exportBackup(){
    const blob=new Blob([JSON.stringify(state,null,2)],{type:'application/json'});const url=URL.createObjectURL(blob);const a=document.createElement('a');a.href=url;a.download=`spell-garden-backup-${new Date().toISOString().slice(0,10)}.json`;a.click();setTimeout(()=>URL.revokeObjectURL(url),1000);
  }

  }
