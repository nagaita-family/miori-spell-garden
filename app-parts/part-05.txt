  function checkChoice(button,isCorrect,attempt,word,task){
    if (button.disabled) return;
    task.tries++;
    recordAttempt(word,attempt,isCorrect,task.tries===1,task.stage);
    if (isCorrect){
      $$('.choice-btn',$('#questionArea')).forEach(b=>b.disabled=true);button.classList.add('correct');
      handleCorrect(task,word);
    }else{
      button.classList.add('wrong');button.disabled=true;handleWrong(task,word,attempt);
    }
  }

  function checkTyped(input,isCorrect,attempt,word,task,expected){
    if (!attempt) return;
    task.tries++;
    recordAttempt(word, task.stage===3 ? word.word.slice(0,task.span.start)+attempt+word.word.slice(task.span.end+1) : attempt, isCorrect, task.tries===1, task.stage);
    if(isCorrect){input.disabled=true;handleCorrect(task,word)}
    else{
      input.classList.add('wrong'); setTimeout(()=>input.classList.remove('wrong'),350);
      showLetterFeedback(attempt,expected); handleWrong(task,word,attempt); input.select();
    }
  }

  function showLetterFeedback(attempt,expected){
    const n=Math.max(attempt.length,expected.length);let html='<div class="letter-feedback">';
    for(let i=0;i<n;i++){const a=attempt[i]||'·', e=expected[i]||'';html+=`<span class="${a===e?'ok':'bad'}">${escapeHtml(a)}</span>`}html+='</div>';
    $('#feedback').innerHTML=html;
  }

  function handleCorrect(task,word){
    const base=[8,10,12,18][task.stage-1];
    const bonus=task.tries===1?4:0;
    const gained=base+bonus;
    session.xp+=gained;if(task.tries===1)session.firstTryWins++;
    $('#sessionXp').textContent=`+${session.xp} XP`;
    $('#feedback').className='feedback good';
    if(!$('#feedback').querySelector('.letter-feedback')) $('#feedback').textContent=task.tries===1?'Perfect — you remembered it! ✦':'Yes! You fixed it. That counts. ♡';
    playSfx('correct');
    setTimeout(()=>{session.index++;renderTask()},780);
  }

  function handleWrong(task,word){
    $('#feedback').classList.add('bad');

    playSfx('wrong');
    if(task.tries>=2) setTimeout(()=>showHint(),300);
  }

  }

  function showHint(){
    const task=currentTask(),word=currentWord();if(!task||!word)return;
    task.usedHint=true;word.learning.hintUsed++;saveStateQuietly();
    playWordAudio(word,true);
    let target;
    if(task.stage===4){
      const inp=$('.full-input');const typed=(inp?.value||'').toLowerCase();let idx=0;while(idx<word.word.length&&typed[idx]===word.word[idx])idx++;idx=Math.min(idx,word.word.length-1);target={index:idx,char:word.word[idx]};
    }else{const span=task.span||weakSpanFor(word);target={index:span.start,char:word.word[span.start]}}
    const choices=shuffle([target.char,...closestConfusion(target.char)]).slice(0,3);
    const panel=$('#helpPanel');
    panel.innerHTML=`<strong>Hint:</strong> listen slowly. Try this letter near the tricky spot.<div class="hint-choices">${choices.map(c=>`<button class="hint-choice" data-char="${escapeHtml(c)}">${escapeHtml(c)}</button>`).join('')}</div>`;
    panel.classList.remove('hidden');
    $$('.hint-choice',panel).forEach(b=>b.onclick=()=>{
      if(b.dataset.char!==target.char){b.classList.add('wrong');return}
      b.classList.add('correct');insertHintCharacter(task,word,target.index,target.char);setTimeout(()=>panel.classList.add('hidden'),450);
    });
  }

  function insertHintCharacter(task,word,index,char){
    if(task.stage===3){
      const input=$('.gap-input');const span=task.span;const local=Math.max(0,index-span.start);let arr=(input.value||'').split('');arr[local]=char;input.value=arr.join('');input.focus();
    } else if(task.stage===4){
      const input=$('.full-input');let arr=(input.value||'').split('');while(arr.length<index)arr.push('');arr[index]=char;input.value=arr.join('');input.focus();
    }
  }
